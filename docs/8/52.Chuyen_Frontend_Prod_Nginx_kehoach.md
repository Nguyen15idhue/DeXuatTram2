### Yêu cầu cần đạt
```bash
cd /root/DeXuatTram2
mkdir -p /root/backups
cp .env .env.bak.$(date +%Y%m%d_%H%M)
# 1) Dump DB
docker compose -f docker-compose.simple.yml exec -T mysql sh -c 'mysqldump -uroot -p"$MYSQL_ROOT_PASSWORD" --default-character-set=utf8mb4 "$MYSQL_DATABASE"' > /root/backups/dexuat_pre52_$(date +%Y%m%d_%H%M).sql
# 2) Snapshot uploads THẬT (tar ra /root/backups)
docker run --rm -v dexuattram2_uploads_data:/d -v /root/backups:/b busybox tar czf /b/uploads_pre52_$(date +%Y%m%d_%H%M).tar.gz -C /d .
# 3) Ghi trạng thái
docker ps --format 'table {{.Names}}\t{{.Status}}' | grep station
# 4) Xác nhận cả 2 file backup > 0
ls -lh /root/backups
```
