### Logic assign/remove field
```js
const assignFieldToCell = (fieldId, rowIndex, colIndex) => {
  // Update form_fields: set rowIndex, colIndex cho field
  setAssignedFields(prev => prev.map(f => 
    f.field_id === fieldId 
      ? { ...f, config: { ...f.config, rowIndex, colIndex } }
      : f
  ));
};

const getCellField = (rowIndex, colIndex) => {
  return assignedFields.find(f => 
    f.config?.rowIndex === rowIndex && f.config?.colIndex === colIndex
  );
};
```

const handleSave = async () => {

        config: field.config
      }, token);

## Bước 8: DynamicForm — Layout rendering

**File:** `frontend/src/components/dynamic/DynamicForm.jsx`

Sửa phần render (thay thế lines 316-331):

```jsx
// Đọc layout_config
const layoutConfig = formConfig?.layout_config;
const rows = layoutConfig?.rows || [];

// Map field_id -> field config
const fieldMap = {};
fields.forEach(f => { fieldMap[f.field_id] = f; });

// Tách assigned vs unassigned fields
const assignedFields = fields.filter(f => f.config?.rowIndex !== undefined);
const unassignedFields = fields.filter(f => f.config?.rowIndex === undefined);

return (
  <form className="dynamic-form" onSubmit={handleSubmit}>
    {error && <div className="error-message">{error}</div>}
    
    {/* Render theo layout rows */}
    {rows.map((row, rowIdx) => {
      const colCount = parseInt(row.columns.split(':')[1]);
      const rowFields = assignedFields.filter(f => f.config.rowIndex === rowIdx);
      
      return (
        <div key={row.id} className="form-row" data-cols={row.columns}>
          {Array.from({ length: colCount }).map((_, colIdx) => {
            const field = rowFields.find(f => f.config.colIndex === colIdx);
            if (!field) return <div key={colIdx} className="form-cell empty" />;
            if (!evaluateConditions(field, formData)) return <div key={colIdx} className="form-cell hidden" />;
            
            const label = field.config?.labelOverride || field.label;
            const placeholder = field.config?.placeholderOverride || field.placeholder;
            const isRequired = field.config?.requiredOverride ?? field.required;
            const isReadonly = field.config?.readonly || false;
            const rowSpan = field.config?.rowSpan || 1;
            
            return (
              <div 
                key={colIdx} 
                className="form-cell"
                data-rowspan={rowSpan}
              >
                <label>
                  {label}
                  {isRequired && <span style={{ color: '#dc2626' }}> *</span>}
                </label>
                {renderField(field, { placeholder, readonly: isReadonly })}
                {field.help_text && <div className="field-help">{field.help_text}</div>}
                {errors[field.key] && <div className="field-error">{errors[field.key]}</div>}
              </div>
            );
          })}
        </div>
      );
    })}
    
    {/* Fields chưa assigned → render flat */}
    {unassignedFields.map(field => (
      <div key={field.field_id} className="dynamic-form-field">
        <label>
          {field.config?.labelOverride || field.label}
          {(field.config?.requiredOverride ?? field.required) && <span style={{ color: '#dc2626' }}> *</span>}
        </label>
        {renderField(field, { 
          placeholder: field.config?.placeholderOverride || field.placeholder,
          readonly: field.config?.readonly || false
        })}
      </div>
    ))}
    
    {children ? (
      <div className="form-actions">
        {children}
        <button type="submit" className="btn btn-primary">Lưu</button>
      </div>
    ) : (
      <div className="form-actions">
        <button type="submit" className="btn btn-primary">Lưu</button>
      </div>
    )}
  </form>
);
```

## Bước 9: DynamicForm — Conditional visibility

**File:** `frontend/src/components/dynamic/DynamicForm.jsx`

Thêm hàm evaluateConditions:

```jsx
const evaluateConditions = (field, formData) => {
  const conditions = field.config?.conditions || [];
  if (!conditions.length) return true;
  
  const logic = field.config?.conditionLogic || 'AND';
  const results = conditions.map(cond => {
    const val = formData[cond.field];
    switch (cond.operator) {
      case '=': return String(val) === String(cond.value);
      case '!=': return String(val) !== String(cond.value);
      case 'contains': return String(val).includes(cond.value);
      case '>': return Number(val) > Number(cond.value);
      case '<': return Number(val) < Number(cond.value);
      case 'empty': return !val;
      case 'not_empty': return !!val;
      default: return true;
    }
  });
  
  return logic === 'AND' ? results.every(Boolean) : results.some(Boolean);
};
```

## Bước 10: DynamicForm — Field overrides

**File:** `frontend/src/components/dynamic/DynamicForm.jsx`

Sửa hàm renderField để nhận thêm options:

```jsx
const renderField = (field, options = {}) => {
  const { placeholder, readonly } = options;
  const resolvedOptions = getFilteredOptions(field);
  const fieldForRender = { 
    ...field, 
    options: resolvedOptions,
    placeholder: placeholder || field.placeholder
  };

  if (field.type === 'formula') {
    const isPost = field.formula_config?.compute_mode === 'post';
    return (
      <input type="text" className="form-control" value={formData[field.key] || ''} 
             readOnly disabled placeholder={isPost ? 'Tính sau khi lưu' : 'Tính tự động'} />
    );
  }

  if (readonly) {
    return <input type="text" className="form-control" value={formData[field.key] || ''} readOnly />;
  }

  return (
    <DynamicField field={fieldForRender} value={formData[field.key]}
      onChange={(val) => handleChange(field.key, val)} error={errors[field.key]} entityType={entity} />
  );
};
```
