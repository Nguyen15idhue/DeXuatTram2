## SQL Scripts

### Script 1: `database/35-standardize-users.sql`

```sql
-- Thêm field chuc_vu
INSERT INTO field_definitions (entity, key, label, type, source_type, required, status, order_index)
VALUES ('users', 'chuc_vu', 'Chức vụ', 'text', 'json', false, 'active', 50);

-- Tạo data list dm_phong_ban
INSERT INTO data_lists (name, description, columns_config, status) VALUES
('dm_phong_ban', 'Danh mục phòng ban',
 '[{"key":"ma","label":"Mã","type":"text"},{"key":"ten","label":"Tên phòng ban","type":"text"}]',
 'active');

-- Seed rows dm_phong_ban
INSERT INTO data_list_rows (data_list_id, data, status) VALUES
(@dl_id, '{"ma":"KT","ten":"Kỹ thuật"}', 'active'),
(@dl_id, '{"ma":"VH","ten":"Vận hành"}', 'active'),
(@dl_id, '{"ma":"HC","ten":"Hành chính"}', 'active'),
(@dl_id, '{"ma":"KD","ten":"Kinh doanh"}', 'active'),
(@dl_id, '{"ma":"MKT","ten":"Marketing"}', 'active');

-- Link department với dm_phong_ban
```
